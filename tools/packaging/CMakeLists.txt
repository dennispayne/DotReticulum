cmake_minimum_required(VERSION 3.30)
project(DotReticulumInstaller NONE)

foreach(required IN ITEMS DOTRETICULUM_PACKAGE_DIRECTORY DOTRETICULUM_PACKAGE_VERSION
        DOTRETICULUM_PACKAGE_BASENAME DOTRETICULUM_RID)
    if(NOT DEFINED ${required} OR "${${required}}" STREQUAL "")
        message(FATAL_ERROR "${required} is required")
    endif()
endforeach()

file(TO_CMAKE_PATH "${DOTRETICULUM_PACKAGE_DIRECTORY}" DOTRETICULUM_PACKAGE_DIRECTORY)
set(manifest "${DOTRETICULUM_PACKAGE_DIRECTORY}/dotreticulum-package.json")
file(READ "${manifest}" package_json)
string(JSON package_rid GET "${package_json}" rid)
if(NOT "${package_rid}" STREQUAL "${DOTRETICULUM_RID}")
    message(FATAL_ERROR "The staged package RID does not match ${DOTRETICULUM_RID}")
endif()

string(JSON file_count LENGTH "${package_json}" files)
if(file_count LESS 2)
    message(FATAL_ERROR "The staged package does not contain the executable set")
endif()
math(EXPR last_file "${file_count} - 1")
foreach(index RANGE ${last_file})
    string(JSON name GET "${package_json}" files ${index} name)
    if(name MATCHES "[/\\\\]" OR name STREQUAL "." OR name STREQUAL "..")
        message(FATAL_ERROR "Invalid package file name: ${name}")
    endif()
    if(name STREQUAL "LICENSE")
        if(DOTRETICULUM_RID STREQUAL "win-x64")
            install(FILES "${DOTRETICULUM_PACKAGE_DIRECTORY}/${name}" DESTINATION .)
        else()
            install(FILES "${DOTRETICULUM_PACKAGE_DIRECTORY}/${name}" DESTINATION share/doc/dotreticulum)
        endif()
    else()
        install(PROGRAMS "${DOTRETICULUM_PACKAGE_DIRECTORY}/${name}" DESTINATION bin)
    endif()
endforeach()
if(DOTRETICULUM_RID STREQUAL "win-x64")
    install(FILES "${manifest}" DESTINATION .)
else()
    install(FILES "${manifest}" DESTINATION share/doc/dotreticulum)
endif()

set(CPACK_PACKAGE_NAME "DotReticulum")
set(CPACK_PACKAGE_VENDOR "DotReticulum contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Experimental unofficial .NET Reticulum implementation")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/dennispayne/DotReticulum")
set(CPACK_PACKAGE_VERSION "${DOTRETICULUM_PACKAGE_VERSION}")
set(CPACK_PACKAGE_FILE_NAME "${DOTRETICULUM_PACKAGE_BASENAME}")
set(CPACK_PACKAGE_INSTALL_DIRECTORY "DotReticulum")
configure_file("${DOTRETICULUM_PACKAGE_DIRECTORY}/LICENSE" "${CMAKE_CURRENT_BINARY_DIR}/LICENSE.txt" COPYONLY)
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_BINARY_DIR}/LICENSE.txt")

if(DOTRETICULUM_RID STREQUAL "win-x64")
    if(NOT CMAKE_HOST_WIN32 OR NOT DOTRETICULUM_MSI_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
        message(FATAL_ERROR "Windows packaging requires Windows and a three-part MSI version")
    endif()
    set(CPACK_PACKAGE_NAME "DotReticulum ${DOTRETICULUM_PACKAGE_VERSION}")
    set(CPACK_PACKAGE_VERSION "${DOTRETICULUM_MSI_VERSION}")
    set(CPACK_WIX_VERSION 4)
    set(CPACK_WIX_ARCHITECTURE x64)
    set(CPACK_WIX_SIZEOF_VOID_P 8)
    set(CPACK_SYSTEM_NAME win64)
    set(CPACK_WIX_INSTALL_SCOPE perMachine)
    set(CPACK_WIX_UPGRADE_GUID "6C88E8A8-73A8-438A-90FC-83244E24E77B")
    set(CPACK_WIX_PATCH_FILE "${CMAKE_CURRENT_SOURCE_DIR}/WixPathPatch.xml")
    set(CPACK_WIX_PROPERTY_ARPHELPLINK "${CPACK_PACKAGE_HOMEPAGE_URL}")
elseif(DOTRETICULUM_RID STREQUAL "linux-x64")
    if(NOT CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "Linux packaging requires a Linux host")
    endif()
    set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
    set(CPACK_DEBIAN_PACKAGE_MAINTAINER "DotReticulum contributors")
    set(CPACK_DEBIAN_PACKAGE_SECTION "net")
    set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE "amd64")
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_RPM_PACKAGE_NAME "dotreticulum")
    set(CPACK_RPM_PACKAGE_ARCHITECTURE "x86_64")
    set(CPACK_RPM_PACKAGE_LICENSE "Reticulum License")
    set(CPACK_RPM_PACKAGE_GROUP "Applications/Internet")
    # Preserve the published binary bytes recorded in the package manifest.
    set(CPACK_RPM_SPEC_MORE_DEFINE "%define __spec_install_post /bin/true")
elseif(DOTRETICULUM_RID STREQUAL "osx-arm64")
    if(NOT CMAKE_HOST_APPLE)
        message(FATAL_ERROR "macOS packaging requires a macOS host")
    endif()
    set(CPACK_PACKAGING_INSTALL_PREFIX "/usr/local")
    set(CPACK_PRODUCTBUILD_IDENTIFIER "net.dotreticulum.utilities")
    set(CPACK_PRODUCTBUILD_DOMAINS ON)
    set(CPACK_PRODUCTBUILD_DOMAINS_ANYWHERE ON)
    set(CPACK_PRODUCTBUILD_DOMAINS_ROOT ON)
    set(CPACK_PRODUCTBUILD_DOMAINS_USER OFF)
else()
    message(FATAL_ERROR "Unsupported package RID: ${DOTRETICULUM_RID}")
endif()

include(CPack)
